cmake_minimum_required(VERSION 3.10)
project(CaroGameTests LANGUAGES CXX)
enable_testing()
add_executable(caro_tests tests/caro_tests.cpp)
set_target_properties(caro_tests PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
if(MSVC)
    target_compile_options(caro_tests PRIVATE /W4 /WX)
else()
    target_compile_options(caro_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME caro_rules_and_regressions COMMAND caro_tests)
